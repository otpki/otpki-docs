export function ActivationSection({screenshot}) {
  return (
    <>
      <h2>Activation</h2>
      <p>
        The activation section controls whether a key configuration must be
        unlocked before OTPKI can use it.
      </p>
      {screenshot ? (
        <DocScreenshot
          alt="Key configuration activation settings"
          src={screenshot}
        />
      ) : null}
      <h3>Enable Activation</h3>
      <p>
        Use <strong>Enable Activation</strong> when you want OTPKI to require an
        activation code before the key configuration can be used.
      </p>
      <p>
        If this option is turned off, the configuration is treated as always
        active and no activation step is required.
      </p>
      <h3>Auto Activation</h3>
      <p>
        Use <strong>Auto Activation</strong> when you want OTPKI to remember the
        activation code and activate the configuration automatically when the
        application starts.
      </p>
      <p>
        When this is enabled, OTPKI stores the activation code in encrypted form
        for later reuse.
      </p>
      <h3>Verify Activation</h3>
      <p>
        Use <strong>Verify Activation</strong> when you want OTPKI to check that
        the activation code matches a specific value.
      </p>
      <h3>Activation Code To Verify</h3>
      <p>
        Use <strong>Activation Code To Verify</strong> to provide the value that
        OTPKI compares against during activation when verification is enabled.
      </p>
      <p>
        This is mainly useful for key storage backends that do not have their
        own external PIN or unlock check and need OTPKI to enforce the expected
        activation code itself.
      </p>
    </>
  );
}

export function SecuritySection({pluginName, screenshot}) {
  return (
    <>
      <h2>Security</h2>
      <p>
        The security section defines how OTPKI protects sensitive key material
        and backend-specific key data associated with a {pluginName} key
        configuration.
      </p>
      {screenshot ? (
        <DocScreenshot
          alt={`${pluginName} key configuration security settings`}
          src={screenshot}
        />
      ) : null}
      <h3>Encryption Key</h3>
      <p>
        Use <strong>Encryption Key</strong> to select the cryptographic key OTPKI
        will use to encrypt stored key data that must be protected at rest.
      </p>
      <p>
        The selected key must allow <strong>encrypt</strong> and
        <strong>decrypt</strong> usage so OTPKI can protect and later read the
        stored configuration data.
      </p>
      <p>
        If no encryption key is selected, OTPKI falls back to the password
        supplied through the <code>OTPKI_KEY_ENCRYPTION_PASSWORD</code>
        environment variable, supplied during deployment. If that value is blank, the data is encrypted
        using an empty string (<code>""</code>) as the password.
      </p>
    </>
  );
}
